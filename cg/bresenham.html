✏️ Enter Coordinates

X must be between 0–600 and Y between 0–300.

📘 Algorithm

  1. Enter the starting and ending coordinates.
  2. Calculate dx and dy using the absolute coordinate differences.
  3. Determine the directions of x and y.
  4. Initialize the error value to dx − dy.
  5. Plot pixels and update the error until the endpoint is reached.

💻 Java Source Code

import java.awt.*;
import javax.swing.*;

public class Bresenham_Line extends JPanel {
    int x1, y1, x2, y2;

    Bresenham_Line(int x1, int y1, int x2, int y2) {
        this.x1 = x1;
        this.y1 = y1;
        this.x2 = x2;
        this.y2 = y2;
    }

    public void paintComponent(Graphics g) {
        super.paintComponent(g);

        int x = x1, y = y1;
        int dx = Math.abs(x2 - x1);
        int dy = Math.abs(y2 - y1);

        int sx = x1 < x2 ? 1 : -1;
        int sy = y1 < y2 ? 1 : -1;
        int error = dx - dy;

        g.setColor(Color.RED);

        while (true) {
            g.fillRect(x, y, 3, 3);

            if (x == x2 && y == y2) {
                break;
            }

            int e2 = 2 * error;

            if (e2 > -dy) {
                error -= dy;
                x += sx;
            }

            if (e2 < dx) {
                error += dx;
                y += sy;
            }
        }
    }

    public static void main(String[] args) {
        try {
            int x1 = Integer.parseInt(
                JOptionPane.showInputDialog("Enter X1"));
            int y1 = Integer.parseInt(
                JOptionPane.showInputDialog("Enter Y1"));
            int x2 = Integer.parseInt(
                JOptionPane.showInputDialog("Enter X2"));
            int y2 = Integer.parseInt(
                JOptionPane.showInputDialog("Enter Y2"));

            JFrame f = new JFrame("Bresenham Line Drawing");
            f.add(new Bresenham_Line(x1, y1, x2, y2));
            f.setSize(650, 400);
            f.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
            f.setVisible(true);

        } catch (NumberFormatException e) {
            JOptionPane.showMessageDialog(
                null, "Enter valid integer coordinates.");
        }
    }
}
