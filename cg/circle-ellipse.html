🎯 Objective

To implement the Midpoint Circle and Midpoint Ellipse drawing algorithms using integer decision parameters and symmetry, and to observe their graphical output.

1. Midpoint Circle Drawing Algorithm

Algorithm

  1. Read the circle centre (xc, yc) and radius r.
  2. Initialize x = 0, y = r and p = 1 − r.
  3. Plot eight symmetric points for every calculated point.
  4. If p is less than 0, update the decision parameter for the next x.
  5. Otherwise, decrease y and update the decision parameter.
  6. Repeat while x is less than or equal to y.

2. Midpoint Ellipse Drawing Algorithm

Algorithm

  1. Read the centre (xc, yc), horizontal radius rx and vertical radius ry.
  2. Start at x = 0 and y = ry.
  3. Calculate the initial decision parameter for Region 1.
  4. Plot four symmetric points around the centre.
  5. Continue Region 1 while the ellipse slope is less than 1.
  6. Calculate the Region 2 decision parameter and continue until y reaches 0.

💻 Java Source Code

This Java Swing program draws both shapes using their respective midpoint algorithms. Save the source separately as CircleEllipse.java. 

import javax.swing.*;
import java.awt.*;

public class CircleEllipse extends JPanel {

    public void paintComponent(Graphics g) {
        super.paintComponent(g);

        setBackground(Color.WHITE);

        g.setFont(new Font("Arial", Font.BOLD, 16));

        g.setColor(Color.BLUE);
        g.drawString("Midpoint Circle", 70, 30);
        drawCircle(g, 150, 150, 80);

        g.setColor(Color.RED);
        g.drawString("Midpoint Ellipse", 350, 30);
        drawEllipse(g, 400, 150, 130, 70);
    }

    // Midpoint Circle Algorithm
    private void drawCircle(Graphics g, int xc, int yc, int r) {
        int x = 0;
        int y = r;
        int p = 1 - r;

        while (x <= y) {
            plotCirclePoints(g, xc, yc, x, y);

            x++;

            if (p < 0) {
                p += 2 * x + 1;
            } else {
                y--;
                p += 2 * (x - y) + 1;
            }
        }
    }

    private void plotCirclePoints(
            Graphics g, int xc, int yc, int x, int y) {

        g.fillRect(xc + x, yc + y, 2, 2);
        g.fillRect(xc - x, yc + y, 2, 2);
        g.fillRect(xc + x, yc - y, 2, 2);
        g.fillRect(xc - x, yc - y, 2, 2);

        g.fillRect(xc + y, yc + x, 2, 2);
        g.fillRect(xc - y, yc + x, 2, 2);
        g.fillRect(xc + y, yc - x, 2, 2);
        g.fillRect(xc - y, yc - x, 2, 2);
    }

    // Midpoint Ellipse Algorithm
    private void drawEllipse(
            Graphics g, int xc, int yc, int rx, int ry) {

        long rx2 = (long) rx * rx;
        long ry2 = (long) ry * ry;

        long x = 0;
        long y = ry;

        long dx = 0;
        long dy = 2 * rx2 * y;

        double p1 = ry2 - rx2 * ry + 0.25 * rx2;

        while (dx < dy) {
            plotEllipsePoints(g, xc, yc, (int) x, (int) y);

            x++;
            dx += 2 * ry2;

            if (p1 < 0) {
                p1 += dx + ry2;
            } else {
                y--;
                dy -= 2 * rx2;
                p1 += dx - dy + ry2;
            }
        }

        double p2 = ry2 * (x + 0.5) * (x + 0.5)
                  + rx2 * (y - 1.0) * (y - 1.0)
                  - rx2 * ry2;

        while (y >= 0) {
            plotEllipsePoints(g, xc, yc, (int) x, (int) y);

            y--;
            dy -= 2 * rx2;

            if (p2 > 0) {
                p2 += rx2 - dy;
            } else {
                x++;
                dx += 2 * ry2;
                p2 += dx - dy + rx2;
            }
        }
    }

    private void plotEllipsePoints(
            Graphics g, int xc, int yc, int x, int y) {

        g.fillRect(xc + x, yc + y, 2, 2);
        g.fillRect(xc - x, yc + y, 2, 2);
        g.fillRect(xc + x, yc - y, 2, 2);
        g.fillRect(xc - x, yc - y, 2, 2);
    }

    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            JFrame frame = new JFrame("Circle and Ellipse Algorithms");

            frame.add(new CircleEllipse());
            frame.setSize(650, 350);
            frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
            frame.setLocationRelativeTo(null);
            frame.setVisible(true);
        });
    }
}
