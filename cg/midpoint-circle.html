The complete circle must fit inside the canvas.

📘 Algorithm

  1. Enter the centre coordinates and radius.
  2. Initialize x = 0, y = r and p = 1 − r.
  3. Plot eight symmetric points.
  4. Increase x and update the decision parameter.
  5. Decrease y when required and repeat while x ≤ y.

💻 Java Source Code

import java.awt.*;
import javax.swing.*;

public class Midpoint_Circle extends JPanel {
    int xc, yc, r;

    Midpoint_Circle(int xc, int yc, int r) {
        this.xc = xc;
        this.yc = yc;
        this.r = r;
    }

    public void paintComponent(Graphics g) {
        super.paintComponent(g);

        int x = 0;
        int y = r;
        int p = 1 - r;

        g.setColor(Color.GREEN.darker());

        while (x <= y) {
            plot8(g, x, y);
            x++;

            if (p < 0) {
                p += 2 * x + 1;
            } else {
                y--;
                p += 2 * (x - y) + 1;
            }
        }
    }

    void plot8(Graphics g, int x, int y) {
        g.fillRect(xc+x, yc+y, 3, 3);
        g.fillRect(xc-x, yc+y, 3, 3);
        g.fillRect(xc+x, yc-y, 3, 3);
        g.fillRect(xc-x, yc-y, 3, 3);

        g.fillRect(xc+y, yc+x, 3, 3);
        g.fillRect(xc-y, yc+x, 3, 3);
        g.fillRect(xc+y, yc-x, 3, 3);
        g.fillRect(xc-y, yc-x, 3, 3);
    }

    public static void main(String[] args) {
        try {
            int xc = Integer.parseInt(
                JOptionPane.showInputDialog("Enter Centre X"));
            int yc = Integer.parseInt(
                JOptionPane.showInputDialog("Enter Centre Y"));
            int r = Integer.parseInt(
                JOptionPane.showInputDialog("Enter Radius"));

            if (r <= 0) {
                JOptionPane.showMessageDialog(
                    null, "Radius must be greater than zero.");
                return;
            }

            JFrame f = new JFrame("Midpoint Circle Drawing");
            f.add(new Midpoint_Circle(xc, yc, r));
            f.setSize(650, 400);
            f.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
            f.setVisible(true);

        } catch (NumberFormatException e) {
            JOptionPane.showMessageDialog(
                null, "Enter valid integer values.");
        }
    }
}
